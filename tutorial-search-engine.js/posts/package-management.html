Package Management: apt & winget

apt (Debian/Ubuntu)

Basic Commands

# Update package list
sudo apt update

# Upgrade all packages
sudo apt upgrade

# Upgrade with package removals if needed
sudo apt full-upgrade

# Install package
sudo apt install package-name
sudo apt install package1 package2 package3

# Install specific version
sudo apt install package=version

# Remove package (keep config files)
sudo apt remove package-name

# Remove package and config files
sudo apt purge package-name

# Remove unused dependencies
sudo apt autoremove

# Clean package cache
sudo apt clean
sudo apt autoclean

Search and Information

# Search for packages
apt search keyword
apt search "exact phrase"

# Show package information
apt show package-name

# List installed packages
apt list --installed

# List upgradable packages
apt list --upgradable

# Check which package provides a file
apt-file search /path/to/file
dpkg -S /path/to/file

# Show package dependencies
apt depends package-name

# Show reverse dependencies
apt rdepends package-name

Package Management

# Download package without installing
apt download package-name

# Install local .deb file
sudo apt install ./package.deb
sudo dpkg -i package.deb

# Fix broken dependencies
sudo apt --fix-broken install

# Reconfigure package
sudo dpkg-reconfigure package-name

# Hold package (prevent upgrades)
sudo apt-mark hold package-name

# Unhold package
sudo apt-mark unhold package-name

# List held packages
apt-mark showhold

Repository Management

# Add repository (Ubuntu)
sudo add-apt-repository ppa:owner/repo-name
sudo apt update

# Remove repository
sudo add-apt-repository --remove ppa:owner/repo-name

# List repositories
cat /etc/apt/sources.list
ls /etc/apt/sources.list.d/

# Add GPG key
wget -qO - https://example.com/key.gpg | sudo apt-key add -

# Or with new method
wget -qO - https://example.com/key.gpg | sudo gpg --dearmor -o /usr/share/keyrings/example.gpg

Common Tasks

# Update and upgrade system
sudo apt update && sudo apt upgrade -y

# Install build essentials
sudo apt install build-essential

# Install development tools
sudo apt install git curl wget vim

# Install web server stack
sudo apt install nginx mysql-server php-fpm

# Install Python development
sudo apt install python3 python3-pip python3-venv

# Install Node.js (via NodeSource)
curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -
sudo apt install nodejs

Troubleshooting

# Fix broken packages
sudo apt --fix-broken install
sudo dpkg --configure -a

# Clear cache and retry
sudo apt clean
sudo apt update

# Resolve lock errors
sudo rm /var/lib/apt/lists/lock
sudo rm /var/cache/apt/archives/lock
sudo rm /var/lib/dpkg/lock*
sudo dpkg --configure -a

# Check for held packages
apt-mark showhold

# Verify package integrity
sudo apt-get check

yum/dnf (RHEL/CentOS/Fedora)

dnf is the modern replacement for yum. Commands are mostly interchangeable.

Basic Commands

# Update all packages
sudo dnf check-update
sudo dnf upgrade

# Install package
sudo dnf install package-name

# Remove package
sudo dnf remove package-name

# Clean up
sudo dnf autoremove
sudo dnf clean all

Search and Information

# Search for a package
sudo dnf search keyword

# Show package info
dnf info package-name

# List installed packages
dnf list installed

pacman (Arch Linux)

Basic Commands

# Sync package databases and upgrade
sudo pacman -Syu

# Install package
sudo pacman -S package-name

# Remove package
sudo pacman -R package-name

# Remove package and dependencies
sudo pacman -Rs package-name

# Clean cache
sudo pacman -Scc

Search and Information

# Search for a package
pacman -Ss keyword

# Show package info
pacman -Si package-name

# List installed packages
pacman -Q

Homebrew (macOS/Linux)

Basic Commands

# Update Homebrew and formulae
brew update

# Upgrade all packages
brew upgrade

# Install package (formula)
brew install package-name

# Install application (cask)
brew install --cask application-name

# Uninstall package
brew uninstall package-name

# Clean up old versions
brew cleanup

Search and Information

# Search for a formula or cask
brew search text

# Show package info
brew info package-name

# List installed packages
brew list

winget (Windows)

Basic Commands

# Search for packages
winget search "package name"
winget search --name "exact name"

# Show package information
winget show package-id

# Install package
winget install package-id
winget install Microsoft.PowerShell

# Install with specific version
winget install package-id --version 1.2.3

# Install multiple packages
winget install package1 package2 package3

# Install interactively
winget install package-id --interactive

# Uninstall package
winget uninstall package-id

# Upgrade specific package
winget upgrade package-id

# Upgrade all packages
winget upgrade --all

List and Search

# List installed packages
winget list

# List installed packages from winget source
winget list --source winget

# Search by name
winget search "Visual Studio Code"

# Search by ID
winget search --id Microsoft.VisualStudioCode

# Show package details
winget show Microsoft.VisualStudioCode

# List available versions
winget show package-id --versions

Package Management

# Upgrade all packages
winget upgrade --all

# Upgrade with auto-accept
winget upgrade --all --accept-package-agreements --accept-source-agreements

# Export installed packages
winget export -o packages.json

# Import packages
winget import -i packages.json

# Download package without installing
winget download package-id

# Install with custom location
winget install package-id --location "C:\Custom\Path"

Configuration

# Open settings
winget settings

# Example settings.json
{
  "experimentalFeatures": {
    "experimentalCmd": true,
    "experimentalArg": true
  },
  "installBehavior": {
    "preferences": {
      "scope": "user"
    }
  },
  "source": {
    "autoUpdateIntervalInMinutes": 5
  }
}

Package Manifest

Export installed packages to reinstall on new machine:

# Export to JSON
winget export -o installed-packages.json

# Import on new machine
winget import -i installed-packages.json --accept-package-agreements --accept-source-agreements

# Example JSON structure
{
  "Sources": [
    {
      "Packages": [
        {
          "PackageIdentifier": "Microsoft.VisualStudioCode"
        },
        {
          "PackageIdentifier": "Git.Git"
        }
      ],
      "SourceDetails": {
        "Name": "winget"
      }
    }
  ]
}

Troubleshooting

# Update winget sources
winget source update

# List sources
winget source list

# Reset sources
winget source reset --force

# Check for updates to winget itself
# winget updates via Microsoft Store

# Verbose output for debugging
winget install package-id --verbose

# Check available upgrades
winget upgrade

# Fix common issues by resetting
winget source reset winget

Best Practices

  • Always update package list before installing (apt update)
  • Regularly upgrade packages for security updates
  • Use autoremove to clean up unused dependencies
  • Export package lists before major system changes
  • Read package descriptions before installing
  • Use specific package IDs with winget for accuracy
  • Check package authenticity and sources
  • Keep a list of critical packages for quick reinstall

Practice Checklist

  • Refresh package metadata before installing.
  • Review dependencies and available versions.
  • Prefer signed packages from trusted repositories.
  • Remove unused packages after testing.
  • Record system changes for future maintenance.
