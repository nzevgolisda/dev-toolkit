TypeScript Guide

Setup

# Install TypeScript
npm install -g typescript

# Initialize project
npm init -y
npm install --save-dev typescript @types/node

# Create tsconfig.json
tsc --init

# Compile
tsc

tsc --watch

Basic Types

// Primitives
let isDone: boolean = false;
let count: number = 42;
let name: string = "Alice";

// Arrays
let numbers: number[] = [1, 2, 3];
let names: Array = ["Alice", "Bob"];

// Tuples
let tuple: [string, number] = ["age", 30];

// Any (avoid when possible)
let anything: any = "hello";

// Unknown (safer than any)
let uncertain: unknown = 4;
if (typeof uncertain === "number") {
  console.log(uncertain + 1);
}

// Void
function log(message: string): void {
  console.log(message);
}

// Never (functions that never return)
function error(message: string): never {
  throw new Error(message);
}

Interfaces

// Basic interface
interface User {
  id: number;
  name: string;
  email: string;
  age?: number;  // Optional property
  readonly created: Date;  // Readonly
}

const user: User = {
  id: 1,
  name: "Alice",
  email: "alice@example.com",
  created: new Date()
};

// Extending interfaces
interface Admin extends User {
  role: string;
  permissions: string[];
}

// Function types
interface SearchFunc {
  (source: string, subString: string): boolean;
}

const search: SearchFunc = (src, sub) => {
  return src.indexOf(sub) > -1;
};

Generics

// Generic function
function identity(arg: T): T {
  return arg;
}

let output = identity("hello");

// Generic interface
interface GenericIdentityFn {
  (arg: T): T;
}

// Generic class
class GenericNumber {
  zeroValue: T;
  add: (x: T, y: T) => T;
}

// Generic constraints
interface Lengthwise {
  length: number;
}

function logLength(arg: T): T {
  console.log(arg.length);
  return arg;
}

// Multiple type parameters
function merge(obj1: T, obj2: U): T & U {
  return { ...obj1, ...obj2 };
}

Classes

class Animal {
  private name: string;
  protected age: number;
  public species: string;

  constructor(name: string, age: number, species: string) {
    this.name = name;
    this.age = age;
    this.species = species;
  }

  // Getter
  get getName(): string {
    return this.name;
  }

  // Setter
  set setAge(age: number) {
    if (age > 0) {
      this.age = age;
    }
  }

  // Method
  makeSound(): void {
    console.log(`${this.name} makes a sound`);
  }
}

// Inheritance
class Dog extends Animal {
  breed: string;

  constructor(name: string, age: number, breed: string) {
    super(name, age, "Canine");
    this.breed = breed;
  }

  makeSound(): void {
    console.log(`${this.getName} barks`);
  }
}

// Abstract class
abstract class Shape {
  abstract getArea(): number;

  printArea(): void {
    console.log(`Area: ${this.getArea()}`);
  }
}

Decorators (Experimental)

Decorators are a special kind of declaration that can be attached to a class declaration, method, accessor, property, or parameter. You need to enable experimentalDecorators in tsconfig.json.

// Class decorator
function sealed(constructor: Function) {
  Object.seal(constructor);
  Object.seal(constructor.prototype);
}

@sealed
class Greeter {
  greeting: string;
  constructor(message: string) {
    this.greeting = message;
  }
  greet() {
    return "Hello, " + this.greeting;
  }
}

// Method decorator
function log(target: any, propertyKey: string, descriptor: PropertyDescriptor) {
    const originalMethod = descriptor.value;
    descriptor.value = function (...args: any[]) {
        console.log(`Calling ${propertyKey} with`, args);
        const result = originalMethod.apply(this, args);
        console.log(`Result:`, result);
        return result;
    };
}

class Calculator {
    @log
    add(a: number, b: number): number {
        return a + b;
    }
}

Type Guards

// typeof guard
function padLeft(value: string, padding: string | number) {
  if (typeof padding === "number") {
    return " ".repeat(padding) + value;
  }
  return padding + value;
}

// instanceof guard
class Bird {
  fly() { console.log("Flying"); }
}
class Fish {
  swim() { console.log("Swimming"); }
}

function move(animal: Bird | Fish) {
  if (animal instanceof Bird) {
    animal.fly();
  } else {
    animal.swim();
  }
}

// Custom type guard
interface Cat {
  meow: () => void;
}
interface Dog {
  bark: () => void;
}

function isCat(pet: Cat | Dog): pet is Cat {
  return (pet as Cat).meow !== undefined;
}

function makeSound(pet: Cat | Dog) {
  if (isCat(pet)) {
    pet.meow();
  } else {
    pet.bark();
  }
}

Advanced Types

Best Practices

  • Enable strict mode in tsconfig.json
  • Use interfaces for objects, type aliases for unions/primitives
  • Avoid using any, use unknown instead
  • Prefer const assertions for literal types
  • Use type guards for narrowing
  • Leverage utility types instead of creating manual types
  • Use readonly for immutable data
  • Define explicit return types for public functions

Safe API Data

Keep external data unknown until a small type guard proves its shape.

interface UserResponse {
  id: number;
  name: string;
}

function isUserResponse(value: unknown): value is UserResponse {
  if (typeof value !== "object" || value === null) return false;
  const user = value as Record<string, unknown>;
  return typeof user.id === "number" && typeof user.name === "string";
}

async function getUser(id: number): Promise<UserResponse> {
  const response = await fetch(`/api/users/${id}`);
  const data: unknown = await response.json();

  if (!response.ok || !isUserResponse(data)) {
    throw new Error("Unexpected user response");
  }
  return data;
}

getUser(42)
  .then(user => console.log(user.name))
  .catch(error => console.error(error));

Practice Checklist

  • Turn on strict checking before adding abstractions.
  • Model external values as unknown.
  • Let the compiler expose unsafe assumptions.
  • Prefer small type guards over broad assertions.
  • Keep generated output out of source control.
