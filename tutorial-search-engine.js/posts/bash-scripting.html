Bash Scripting Guide

Script Basics

#!/bin/bash
# This is a comment

# Make script executable:
# chmod +x script.sh

# Run script:
# ./script.sh

echo "Hello, World!"

Variables

# Define variables (no spaces around =)
name="Alice"
age=30

# Use variables
echo "Name: $name"
echo "Age: ${age}"

# Command substitution
current_date=$(date)
files=$(ls)

# Read user input
read -p "Enter your name: " user_name
echo "Hello, $user_name"

Conditionals

# If statement
if [ "$age" -gt 18 ]; then
    echo "Adult"
elif [ "$age" -eq 18 ]; then
    echo "Just turned adult"
else
    echo "Minor"
fi

# File checks
if [ -f "file.txt" ]; then
    echo "File exists"
fi

if [ -d "directory" ]; then
    echo "Directory exists"
fi

# String comparison
if [ "$name" = "Alice" ]; then
    echo "Name is Alice"
fi

# Multiple conditions
if [ "$age" -gt 18 ] && [ "$name" = "Alice" ]; then
    echo "Adult named Alice"
fi

Comparison Operators

  • -eq - Equal to (numbers)
  • -ne - Not equal to
  • -gt - Greater than
  • -lt - Less than
  • -ge - Greater than or equal
  • -le - Less than or equal
  • = - Equal (strings)
  • != - Not equal (strings)

Loops

# For loop
for i in 1 2 3 4 5; do
    echo "Number: $i"
done

# For loop with range
for i in {1..5}; do
    echo "Count: $i"
done

# Loop over files
for file in *.txt; do
    echo "Processing $file"
done

# While loop
counter=0
while [ $counter -lt 5 ]; do
    echo "Counter: $counter"
    ((counter++))
done

# Until loop
counter=0
until [ $counter -ge 5 ]; do
    echo "Counter: $counter"
    ((counter++))
done

Functions

# Define function
greet() {
    echo "Hello, $1!"
}

# Call function
greet "Alice"

# Function with return value
add() {
    local result=$(($1 + $2))
    echo $result
}

sum=$(add 5 3)
echo "Sum: $sum"

# Function with multiple parameters
process_file() {
    local filename=$1
    local operation=$2
    echo "Processing $filename with $operation"
}

process_file "data.txt" "sort"

Arrays

# Define array
fruits=("apple" "banana" "orange")

# Access elements
echo ${fruits[0]}  # apple
echo ${fruits[1]}  # banana

# All elements
echo ${fruits[@]}

# Array length
echo ${#fruits[@]}

# Loop through array
for fruit in "${fruits[@]}"; do
    echo "Fruit: $fruit"
done

# Add to array
fruits+=("grape")

Command Line Arguments

#!/bin/bash

# $0 - Script name
# $1, $2, ... - Arguments
# $# - Number of arguments
# $@ - All arguments

echo "Script name: $0"
echo "First argument: $1"
echo "Number of arguments: $#"

# Check if argument provided
if [ $# -eq 0 ]; then
    echo "No arguments provided"
    exit 1
fi

# Loop through all arguments
for arg in "$@"; do
    echo "Argument: $arg"
done

File Operations

# Check if file exists
if [ -f "file.txt" ]; then
    echo "File exists"
fi

# Check if directory exists
if [ -d "mydir" ]; then
    echo "Directory exists"
fi

# Create directory
mkdir -p /path/to/directory

# Read file line by line
while IFS= read -r line; do
    echo "Line: $line"
done < file.txt

# Write to file
echo "Content" > file.txt    # Overwrite
echo "More" >> file.txt      # Append

Error Handling

# Exit on error
set -e

# Exit on undefined variable
set -u

# Exit on pipe failure
set -o pipefail

# Combined
set -euo pipefail

# Custom error handling
command || {
    echo "Command failed"
    exit 1
}

# Check exit status
if ! command; then
    echo "Command failed"
    exit 1
fi

Advanced Topics

Command-line Option Parsing with getopts

#!/bin/bash

# A POSIX variable
OPTIND=1         # Reset in case getopts has been used previously in the shell.

# Initialize our own variables:
output_file=""
verbose=0

while getopts "h?vf:" opt; do
    case "$opt" in
    h|\?)
        echo "Usage: $(basename $0) [-h] [-v] [-f output_file]"
        exit 0
        ;;
    v)  verbose=1
        ;;
    f)  output_file=$OPTARG
        ;;
    esac
done

shift $((OPTIND-1))

[ "${1:-}" = "--" ] && shift

echo "verbose=${verbose}, output_file='${output_file}', Leftovers: $@"

# And the rest of the script...

Signal Handling with trap

#!/bin/bash

# Function to be executed when the script is interrupted
cleanup() {
    echo "Caught signal... cleaning up."
    # Add cleanup tasks here
    rm -f /tmp/myscript.*
    exit 1
}

# Trap the INT (Ctrl+C) and TERM signals
trap cleanup INT TERM

echo "Script is running... Press Ctrl+C to interrupt."

# Simulate a long-running process
while true; do
    sleep 1
done

Process Substitution

# Compare the output of two commands without using temporary files
diff <(ls -l /) <(ls -l /home)

Useful Patterns

Logging

log() {
    echo "[$(date +'%Y-%m-%d %H:%M:%S')] $*"
}

log "Starting script"
log "Processing data"

Progress Bar

for i in {1..100}; do
    echo -ne "\rProgress: $i%"
    sleep 0.1
done
echo ""

Confirm Prompt

confirm() {
    read -p "Are you sure? (y/n) " -n 1 -r
    echo
    [[ $REPLY =~ ^[Yy]$ ]]
}

if confirm; then
    echo "Confirmed"
else
    echo "Cancelled"
fi

Practical Examples

Simple Backup Script

#!/bin/bash
set -euo pipefail

SRC_DIR="/path/to/source"
DEST_DIR="/path/to/backup"
TIMESTAMP=$(date +"%Y-%m-%d_%H-%M-%S")
BACKUP_FILE="$DEST_DIR/backup-$TIMESTAMP.tar.gz"

echo "Starting backup of $SRC_DIR to $BACKUP_FILE"
tar -czf "$BACKUP_FILE" "$SRC_DIR"
echo "Backup complete."

System Monitoring Script

#!/bin/bash
set -euo pipefail

LOG_FILE="/var/log/system-monitor.log"

log() {
    echo "[$(date)] $1" >> "$LOG_FILE"
}

check_disk_usage() {
    log "Checking disk usage..."
    df -h
}

check_memory_usage() {
    log "Checking memory usage..."
    free -h
}

log "--- System Monitor Started ---"
check_disk_usage
check_memory_usage
log "--- System Monitor Finished ---"

Best Practices

  • Always quote variables: "$var"
  • Use #!/bin/bash shebang
  • Add comments for complex logic
  • Check if commands exist before using
  • Use set -e for error handling
  • Make scripts idempotent when possible
  • Validate input parameters

Practice Checklist

  • Quote variables unless word splitting is intentional.
  • Use strict modes when the script can fail safely.
  • Check command exit codes at important boundaries.
  • Print useful context for failures.
  • Test with empty, spaced, and unexpected inputs.
