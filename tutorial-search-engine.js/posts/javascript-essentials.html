JavaScript Essentials & Modern Patterns

Variables & Constants

// Use const by default
const name = "Alice";
const age = 30;

// Use let for reassignable variables
let count = 0;
count++;

// Avoid var (function-scoped, hoisted)
// var x = 10; // Don't use this

Arrow Functions

// Traditional function
function add(a, b) {
  return a + b;
}

// Arrow function
const add = (a, b) => a + b;

// Multiple lines
const multiply = (a, b) => {
  const result = a * b;
  return result;
};

// Single parameter
const square = x => x * x;

Destructuring

// Array destructuring
const [first, second, ...rest] = [1, 2, 3, 4, 5];

// Object destructuring
const { name, age } = { name: "Alice", age: 30, city: "NYC" };

// With defaults
const { country = "USA" } = {};

// Renaming
const { name: userName } = { name: "Alice" };

// Function parameters
const greet = ({ name, age }) => {
  console.log(`${name} is ${age}`);
};

Template Literals

const name = "Alice";
const age = 30;

// String interpolation
const greeting = `Hello, ${name}!`;

// Multi-line strings
const message = `
  Name: ${name}
  Age: ${age}
  Year: ${2024 - age}
`;

// Expression evaluation
const result = `2 + 2 = ${2 + 2}`;

Array Methods

const numbers = [1, 2, 3, 4, 5];

// Map - transform each element
const doubled = numbers.map(n => n * 2);

// Filter - select elements
const evens = numbers.filter(n => n % 2 === 0);

// Reduce - accumulate value
const sum = numbers.reduce((acc, n) => acc + n, 0);

// Find - get first match
const found = numbers.find(n => n > 3);

// Some/Every - boolean checks
const hasEven = numbers.some(n => n % 2 === 0);
const allPositive = numbers.every(n => n > 0);

Spread & Rest Operators

// Spread array
const arr1 = [1, 2, 3];
const arr2 = [...arr1, 4, 5]; // [1,2,3,4,5]

// Spread object
const obj1 = { a: 1, b: 2 };
const obj2 = { ...obj1, c: 3 }; // {a:1, b:2, c:3}

// Rest parameters
const sum = (...numbers) => {
  return numbers.reduce((a, b) => a + b, 0);
};

sum(1, 2, 3, 4); // 10

Promises & Async/Await

// Promise
fetch('https://api.example.com/data')
  .then(response => response.json())
  .then(data => console.log(data))
  .catch(error => console.error(error));

// Async/await (preferred)
async function fetchData() {
  try {
    const response = await fetch('https://api.example.com/data');
    const data = await response.json();
    console.log(data);
  } catch (error) {
    console.error(error);
  }
}

// Multiple awaits
async function getMultiple() {
  const [users, posts] = await Promise.all([
    fetch('/api/users').then(r => r.json()),
    fetch('/api/posts').then(r => r.json())
  ]);
}

Object Shorthand

const name = "Alice";

// Property shorthand
const person = { name, age };
// Same as: { name: name, age: age }

// Method shorthand
const obj = {
  greet() {
    return "Hello";
  }
};
// Same as: greet: function() { ... }

Optional Chaining & Nullish Coalescing

// Optional chaining
const user = { profile: { name: "Alice" } };
const userName = user?.profile?.name; // "Alice"
const missing = user?.settings?.theme; // undefined

// Nullish coalescing
const value = null ?? "default"; // "default"
const number = 0 ?? 42; // 0 (only null/undefined trigger ??)

// Combined
const theme = user?.settings?.theme ?? "light";

Classes

class Person {
  constructor(name, age) {
    this.name = name;
    this.age = age;
  }

  greet() {
    return `Hello, I'm ${this.name}`;
  }

  // Static method
  static getInfo() {
    return 'This is a Person class';
  }
}

// Inheritance
class Developer extends Person {
  constructor(name, age, skill) {
    super(name, age);
    this.skill = skill;
  }

  work() {
    return `${this.name} is coding with ${this.skill}`;
  }
}

const alice = new Person('Alice', 30);
const bob = new Developer('Bob', 25, 'React');

Map and Set

// Set - collection of unique values
const mySet = new Set([1, 2, 2, 3, 4, 4]);
mySet.add(5);
mySet.has(3); // true
mySet.delete(1);
console.log(mySet); // Set { 2, 3, 4, 5 }

// Map - collection of key-value pairs
const myMap = new Map();
myMap.set('name', 'Alice');
myMap.set('age', 30);
myMap.get('name'); // 'Alice'
myMap.has('age'); // true

for (const [key, value] of myMap) {
  console.log(`${key}: ${value}`);
}

Error Handling with try...catch...finally

try {
  // Code that might throw an error
  const result = riskyOperation();
  console.log(result);
} catch (error) {
  // Code to handle the error
  console.error('An error occurred:', error.message);
} finally {
  // Code that will always run, regardless of an error
  console.log('Cleanup code here.');
}

The Event Loop (Conceptual)

JavaScript has a single-threaded execution model. The event loop is what allows it to handle asynchronous operations (like `setTimeout`, `fetch`, or user events) without blocking the main thread.

  1. Call Stack: Where functions are executed.
  2. Web APIs: Where async operations live while they are processing.
  3. Callback Queue: Where callback functions wait to be executed after the async operation is complete.
  4. Event Loop: The process that continuously checks if the Call Stack is empty. If it is, it moves the first callback from the Callback Queue to the Call Stack for execution.

Common Patterns

Debounce

function debounce(fn, delay) {
  let timeout;
  return (...args) => {
    clearTimeout(timeout);
    timeout = setTimeout(() => fn(...args), delay);
  };
}

const search = debounce((query) => {
  console.log('Searching for:', query);
}, 300);

Throttle

function throttle(fn, limit) {
  let inThrottle;
  return (...args) => {
    if (!inThrottle) {
      fn(...args);
      inThrottle = true;
      setTimeout(() => inThrottle = false, limit);
    }
  };
}
