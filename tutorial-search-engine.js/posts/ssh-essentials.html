SSH Essentials

Basic Connection

# Connect to server
ssh user@hostname
ssh user@192.168.1.100

# Specify port
ssh -p 2222 user@hostname

# Verbose output (for debugging)
ssh -v user@hostname
ssh -vv user@hostname  # More verbose
ssh -vvv user@hostname  # Maximum verbosity

SSH Key Generation

# Generate new SSH key pair
ssh-keygen -t ed25519 -C "your_email@example.com"

# Or using RSA (if ed25519 not supported)
ssh-keygen -t rsa -b 4096 -C "your_email@example.com"

# Specify custom filename
ssh-keygen -t ed25519 -f ~/.ssh/id_ed25519_work

# Generate with no passphrase (not recommended)
ssh-keygen -t ed25519 -N "" -f ~/.ssh/id_ed25519

Copy Public Key to Server

# Using ssh-copy-id (easiest)
ssh-copy-id user@hostname
ssh-copy-id -i ~/.ssh/id_ed25519.pub user@hostname

# Manual copy
cat ~/.ssh/id_ed25519.pub | ssh user@hostname "mkdir -p ~/.ssh && cat >> ~/.ssh/authorized_keys"

# Or copy manually and paste
cat ~/.ssh/id_ed25519.pub
# Then SSH to server and add to ~/.ssh/authorized_keys

SSH Config File

Create/edit ~/.ssh/config:

# Simple host
Host myserver
    HostName 192.168.1.100
    User admin
    Port 22
    IdentityFile ~/.ssh/id_ed25519

# Multiple configurations
Host work
    HostName work.example.com
    User john
    IdentityFile ~/.ssh/id_rsa_work
    Port 2222

Host github
    HostName github.com
    User git
    IdentityFile ~/.ssh/id_ed25519_github

# Wildcard for multiple servers
Host web*
    User deploy
    IdentityFile ~/.ssh/id_deploy
    StrictHostKeyChecking no

# Jump host (bastion)
Host internal
    HostName 10.0.0.50
    User admin
    ProxyJump bastion

Host bastion
    HostName bastion.example.com
    User admin

# Usage: ssh myserver, ssh work, ssh github

SSH Config Options

Host example
    HostName example.com
    User username
    Port 22
    IdentityFile ~/.ssh/id_rsa
    
    # Connection settings
    ConnectTimeout 10
    ServerAliveInterval 60
    ServerAliveCountMax 3
    
    # Security
    StrictHostKeyChecking yes
    IdentitiesOnly yes
    
    # Forwarding
    ForwardAgent yes
    ForwardX11 yes
    
    # Compression
    Compression yes
    
    # Keep connection alive
    TCPKeepAlive yes

File Transfer

SCP (Secure Copy)

# Copy file to remote
scp file.txt user@host:/path/to/destination

# Copy file from remote
scp user@host:/path/to/file.txt ./

# Copy directory recursively
scp -r directory/ user@host:/path/

# Copy with specific port
scp -P 2222 file.txt user@host:/path/

# Copy with compression
scp -C large_file.tar.gz user@host:/path/

SFTP

# Connect to SFTP server
sftp user@hostname

# SFTP commands
sftp> ls                  # List remote files
sftp> lls                 # List local files
sftp> pwd                 # Remote directory
sftp> lpwd                # Local directory
sftp> cd /path            # Change remote dir
sftp> lcd /path           # Change local dir
sftp> get file.txt        # Download file
sftp> put file.txt        # Upload file
sftp> get -r directory/   # Download directory
sftp> put -r directory/   # Upload directory
sftp> mkdir newdir        # Create remote directory
sftp> rm file.txt         # Delete remote file
sftp> exit                # Quit SFTP

Rsync over SSH

# Sync directory to remote
rsync -avz /local/path/ user@host:/remote/path/

# Sync from remote to local
rsync -avz user@host:/remote/path/ /local/path/

# With progress and delete
rsync -avz --progress --delete src/ user@host:/dest/

# Exclude files
rsync -avz --exclude='*.log' src/ user@host:/dest/

# Dry run (test without changes)
rsync -avz --dry-run src/ user@host:/dest/

Port Forwarding

Local Port Forwarding

# Forward local port to remote
ssh -L 8080:localhost:80 user@host
# Access remote port 80 via localhost:8080

# Forward to different host through SSH server
ssh -L 3306:database.internal:3306 user@bastion
# Access database.internal:3306 via localhost:3306

Remote Port Forwarding

# Forward remote port to local
ssh -R 8080:localhost:3000 user@host
# Remote host can access your localhost:3000 via their localhost:8080

Dynamic Port Forwarding (SOCKS Proxy)

# Create SOCKS proxy
ssh -D 8080 user@host
# Configure browser to use localhost:8080 as SOCKS5 proxy

SSH Agent

# Start SSH agent
eval "$(ssh-agent -s)"

# Add key to agent
ssh-add ~/.ssh/id_ed25519
ssh-add ~/.ssh/id_rsa_work

# List loaded keys
ssh-add -l

# Remove key from agent
ssh-add -d ~/.ssh/id_ed25519

# Remove all keys
ssh-add -D

# Add with timeout (auto-remove after 1 hour)
ssh-add -t 3600 ~/.ssh/id_ed25519

SSH Server Configuration

Edit /etc/ssh/sshd_config:

# Port and listening
Port 22
ListenAddress 0.0.0.0

# Authentication
PermitRootLogin no
PasswordAuthentication no
PubkeyAuthentication yes
AuthorizedKeysFile .ssh/authorized_keys

# Security
AllowUsers user1 user2
DenyUsers baduser
AllowGroups sshusers

# Connection settings
ClientAliveInterval 300
ClientAliveCountMax 2
MaxAuthTries 3
MaxSessions 10

# Subsystem
Subsystem sftp /usr/lib/openssh/sftp-server

# After changes, restart SSH
sudo systemctl restart sshd

Security Best Practices

# Change default port
Port 2222

# Disable root login
PermitRootLogin no

# Disable password authentication (use keys only)
PasswordAuthentication no
ChallengeResponseAuthentication no

# Use strong ciphers
Ciphers chacha20-poly1305@openssh.com,aes256-gcm@openssh.com

# Limit user access
AllowUsers deploy admin

# Use fail2ban to prevent brute force
sudo apt install fail2ban
sudo systemctl enable fail2ban
sudo systemctl start fail2ban

Troubleshooting

# Test connection with verbose output
ssh -vvv user@host

# Check SSH service status
sudo systemctl status sshd

# View SSH logs
sudo tail -f /var/log/auth.log  # Ubuntu/Debian
sudo tail -f /var/log/secure    # RHEL/CentOS

# Test SSH key
ssh-keygen -l -f ~/.ssh/id_ed25519
ssh-keygen -l -f ~/.ssh/id_ed25519.pub

# Check file permissions (important!)
chmod 700 ~/.ssh
chmod 600 ~/.ssh/id_ed25519
chmod 644 ~/.ssh/id_ed25519.pub
chmod 600 ~/.ssh/authorized_keys
chmod 600 ~/.ssh/config

# Verify authorized_keys on server
cat ~/.ssh/authorized_keys

# Test SSH from server to itself
ssh localhost

Remote Development with VS Code

Visual Studio Code's "Remote - SSH" extension allows you to use a remote machine as your full-featured development environment.

  1. Install the Remote - SSH extension in VS Code.
  2. Open the command palette (Ctrl+Shift+P) and select "Remote-SSH: Connect to Host...".
  3. Enter your SSH connection string (e.g., `user@hostname`) or select a host from your `~/.ssh/config` file.
  4. VS Code will open a new window connected to the remote machine, where you can edit files, use the terminal, and debug as if you were working locally.

Persistent Connection

# In ~/.ssh/config
Host *
    ControlMaster auto
    ControlPath ~/.ssh/sockets/%r@%h-%p
    ControlPersist 600

# Create socket directory
mkdir -p ~/.ssh/sockets

Execute Remote Command

# Single command
ssh user@host "ls -la"

# Multiple commands
ssh user@host "cd /var/log && tail -n 20 syslog"

# With sudo
ssh user@host "sudo systemctl restart nginx"

# Redirect output to local file
ssh user@host "cat /var/log/app.log" > local_app.log

Useful Aliases

Add to ~/.bashrc or ~/.zshrc:

# Quick SSH connections
alias ssh-work='ssh work'
alias ssh-prod='ssh production'

# SSH with specific key
alias ssh-github='ssh -i ~/.ssh/id_ed25519_github git@github.com'

# Copy SSH key to clipboard (Linux)
alias sshcopy='cat ~/.ssh/id_ed25519.pub | xclip -selection clipboard'

# Copy SSH key to clipboard (Mac)
alias sshcopy='cat ~/.ssh/id_ed25519.pub | pbcopy'

# Copy SSH key to clipboard (Windows/WSL)
alias sshcopy='cat ~/.ssh/id_ed25519.pub | clip.exe'

  • ssh -D 8080 user@host - SOCKS proxy

Practice Checklist

  • Use keys instead of password-only authentication.
  • Protect private keys with a passphrase.
  • Verify host fingerprints on first connection.
  • Restrict forwarding and login permissions.
  • Review authentication logs after changes.
